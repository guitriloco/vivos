import os
import hashlib

# PROTOCOLO SOMBRA: Ofuscação de Ativos
def ocultar_ativos(diretorio_origem):
    for arquivo in os.listdir(diretorio_origem):
        # Gera um hash único para o nome (Sombra)
        novo_nome = hashlib.md5(arquivo.encode()).hexdigest() + ".dat"
        os.rename(os.path.join(diretorio_origem, arquivo), 
                  os.path.join(diretorio_origem, novo_nome))
        # Log de recuperação (Mantenha este arquivo no Xiaomi 14C via Syncthing)
        with open("mapa_sombra.txt", "a") as log:
            log.write(f"{novo_nome} == {arquivo}\n")

print("Ativos prontos para infiltração no Aether-Sombra.")

import requests
from bs4 import BeautifulSoup
import re

# CONFIGURAÇÃO DE SOBERANIA
ALVO_URL = "URL_DO_REPOSITORIO_ALVO"  # Onde o Néctar está escondido
EXTENSOES_LUXO = ('.pdf', '.epub', '.zip', '.rar', '.mobi')

def minerar_ativos(url):
    print(f"[*] Iniciando Infiltração em: {url}")
    try:
        # Simula um navegador comum para evitar bloqueios (Corporate Streetwear Style)
        headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}
        response = requests.get(url, headers=headers, timeout=10)
        
        if response.status_code == 200:
            soup = BeautifulSoup(response.text, 'html.parser')
            links_encontrados = []

            # Varre todos os links <a> da página
            for link in soup.find_all('a', href=True):
                href = link['href']
                
                # Filtra apenas o que é Ativo de Luxo (Extensões definidas)
                if href.lower().endswith(EXTENSOES_LUXO):
                    # Transforma link relativo em absoluto se necessário
                    link_completo = href if href.startswith('http') else url + href
                    links_encontrados.append(link_completo)

            return list(set(links_encontrados)) # Remove duplicatas na fonte
    except Exception as e:
        print(f"[!] Glitch na Operação: {e}")
        return []

# EXECUÇÃO E GERAÇÃO DO LOG PARA O TERABOX
